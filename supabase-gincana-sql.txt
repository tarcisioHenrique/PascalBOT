-- SQL para criar as tabelas da Gincana Linguística no Supabase
-- Cole este conteúdo no SQL Editor do Supabase e execute.
--
-- Observação importante:
-- As tabelas abaixo cobrem o fluxo já integrado ao Supabase no código:
-- partidas, times, rounds, respostas, eventos, placar e avaliação do Júri.
-- Algumas partes do Clash ainda usam localStorage no frontend atualmente.
-- As tabelas do Clash abaixo já deixam o banco preparado para a próxima etapa
-- de integração multiusuário.

create extension if not exists pgcrypto;

create table if not exists public.gincana_partidas (
  id uuid primary key default gen_random_uuid(),
  nome text not null,
  status text not null default 'aguardando',
  dinamica_atual text not null default 'juri_popular',
  round_atual integer not null default 0,
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now()
);

create table if not exists public.gincana_times (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  nome text not null,
  slug text not null,
  pontos numeric not null default 0,
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now(),
  unique (partida_id, slug)
);

create table if not exists public.gincana_rounds (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  dinamica text not null default 'juri_popular',
  numero integer not null,
  status text not null default 'em_andamento',
  tarefa text not null,
  papeis jsonb not null default '{}'::jsonb,
  resultado_pascal jsonb,
  vencedor_time_id uuid references public.gincana_times(id) on delete set null,
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now(),
  finalizado_em timestamptz,
  unique (partida_id, numero)
);

create table if not exists public.gincana_respostas (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  round_id uuid not null references public.gincana_rounds(id) on delete cascade,
  time_id uuid not null references public.gincana_times(id) on delete cascade,
  texto_resposta text not null,
  nota_final numeric,
  feedback_final text not null default '',
  criterios_avaliacao jsonb,
  justificativa_pascal text not null default '',
  status text not null default 'enviada',
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now(),
  unique (round_id, time_id)
);

create table if not exists public.gincana_eventos (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  tipo text not null,
  descricao text not null,
  time_id uuid references public.gincana_times(id) on delete set null,
  pontos numeric not null default 0,
  criado_em timestamptz not null default now()
);

create table if not exists public.gincana_clash_estado (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  aberto boolean not null default false,
  fase integer not null default 0,
  topico_index integer not null default 0,
  campeao_duelista_id uuid,
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now(),
  unique (partida_id)
);

create table if not exists public.gincana_clash_duelistas (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  time_id uuid not null references public.gincana_times(id) on delete cascade,
  nome text not null,
  status text not null default 'ativo',
  maquina_chave text,
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now()
);

create table if not exists public.gincana_clash_duelos (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  fase integer not null,
  ordem integer not null default 0,
  aluno_a_id uuid references public.gincana_clash_duelistas(id) on delete set null,
  aluno_b_id uuid references public.gincana_clash_duelistas(id) on delete set null,
  status text not null default 'aguardando_inicio',
  pergunta text,
  alternativas jsonb not null default '[]'::jsonb,
  alternativa_correta text not null default '',
  resposta_esperada text not null default '',
  topico text not null default '',
  vez_aluno_id uuid references public.gincana_clash_duelistas(id) on delete set null,
  valor_pergunta numeric not null default 1,
  passe_usado boolean not null default false,
  passe_origem_id uuid references public.gincana_clash_duelistas(id) on delete set null,
  repasse_disponivel_para_id uuid references public.gincana_clash_duelistas(id) on delete set null,
  vencedor_id uuid references public.gincana_clash_duelistas(id) on delete set null,
  criado_em timestamptz not null default now(),
  atualizado_em timestamptz not null default now()
);

alter table public.gincana_clash_duelos
  add column if not exists ordem integer not null default 0;

create table if not exists public.gincana_clash_historico (
  id uuid primary key default gen_random_uuid(),
  partida_id uuid not null references public.gincana_partidas(id) on delete cascade,
  duelo_id uuid references public.gincana_clash_duelos(id) on delete cascade,
  aluno_id uuid references public.gincana_clash_duelistas(id) on delete set null,
  tipo text not null,
  texto text not null default '',
  resposta text,
  correta boolean,
  valor_pergunta numeric,
  pontos_ganhos numeric not null default 0,
  feedback text not null default '',
  criado_em timestamptz not null default now()
);

create index if not exists gincana_times_partida_idx
  on public.gincana_times(partida_id);

create index if not exists gincana_rounds_partida_numero_idx
  on public.gincana_rounds(partida_id, numero desc);

create index if not exists gincana_respostas_round_idx
  on public.gincana_respostas(round_id);

create index if not exists gincana_eventos_partida_criado_idx
  on public.gincana_eventos(partida_id, criado_em desc);

create index if not exists gincana_clash_duelistas_partida_time_idx
  on public.gincana_clash_duelistas(partida_id, time_id);

create index if not exists gincana_clash_duelistas_status_idx
  on public.gincana_clash_duelistas(partida_id, status);

create unique index if not exists gincana_clash_duelista_maquina_unica_idx
  on public.gincana_clash_duelistas(partida_id, maquina_chave)
  where maquina_chave is not null;

drop index if exists public.gincana_clash_duelos_partida_fase_idx;
create index gincana_clash_duelos_partida_fase_idx
  on public.gincana_clash_duelos(partida_id, fase, ordem, criado_em);

create index if not exists gincana_clash_duelos_status_idx
  on public.gincana_clash_duelos(partida_id, status);

create index if not exists gincana_clash_historico_duelo_idx
  on public.gincana_clash_historico(duelo_id, criado_em);

-- Trigger simples para atualizar atualizado_em automaticamente.
create or replace function public.set_gincana_atualizado_em()
returns trigger
language plpgsql
as $$
begin
  new.atualizado_em = now();
  return new;
end;
$$;

drop trigger if exists set_gincana_partidas_atualizado_em on public.gincana_partidas;
create trigger set_gincana_partidas_atualizado_em
before update on public.gincana_partidas
for each row execute function public.set_gincana_atualizado_em();

drop trigger if exists set_gincana_times_atualizado_em on public.gincana_times;
create trigger set_gincana_times_atualizado_em
before update on public.gincana_times
